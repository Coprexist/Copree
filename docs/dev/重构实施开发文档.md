```python
        self,
        db,

        dm_session_id: Optional[str] = None,
        content: str = "",

        self,

        group_id: Optional[int] = None,
        dm_session_id: Optional[str] = None,
        limit: int = 50,
        offset: int = 0,
    ) -> List[dict]: ...

    async def set_member_dnd(
        self,
        db,
        member_id: int,
        group_id: int,
        until: Optional[datetime] = None,
        member_type: str = "ai",
    ) -> dict: ...

    @abstractmethod
    async def get_friend_list(self, db, user_id: int) -> List[dict]: ...

    async def get_user_info(self, db, user_id: int) -> dict: ...
```

```python
# app/skills/base.py（新建）
from abc import ABC, abstractmethod
from dataclasses import dataclass, field
from typing import Any, Dict, List, Optional


@dataclass
class ActDecision:
    should_act: bool
    priority: int = 0
    action_type: str = "silent"
    reason: str = ""


@dataclass
class SkillOutput:
    messages_to_send: List[Dict] = field(default_factory=list)
    state_changes: Dict = field(default_factory=dict)
    memory_updates: List[Dict] = field(default_factory=list)
    internal_log: str = ""


class AutonomousSkill(ABC):
    name: str = ""
    description: str = ""
    segment: str = ""
    subscribed_events: List[str] = []

    resource_budget: Dict = {
        "llm_tokens_per_day": 0,
        "messages_per_day": 0,
    }

    @abstractmethod
    async def should_act(self, event: Dict, state: Dict) -> ActDecision: ...

    @abstractmethod
    async def act(self, event: Dict, decision: ActDecision, state: Dict) -> SkillOutput: ...

    async def load_state(self) -> Dict:
        return {}

    async def save_state(self, state: Dict) -> None:
        pass


class AppSkill(AutonomousSkill):
    required_state: Dict = {}

    async def should_act(self, event: Dict, state: Dict) -> ActDecision:
        pass


class StateSkill(AutonomousSkill):
    @abstractmethod
    async def get_state(self, query: Dict) -> Dict: ...

    @abstractmethod
    async def update_state(self, updates: Dict) -> Dict: ...

    async def publish_state_change(self, change: Dict) -> None:
        pass
```

#### SkillEventBus

```python
# app/services/skill_event_bus.py（新建）
from __future__ import annotations
import asyncio
import logging
from typing import Any, Dict, List, Optional
from dataclasses import dataclass, field

from app.services.event_bus import EventBus, event_bus as system_event_bus

logger = logging.getLogger(__name__)


class SkillRegistry:
    """
    技能注册表 — 作为 services 层与 skills 层之间的中介，
    解决循环依赖问题。services 层只依赖 registry 接口，
    不直接依赖 skills 包。
    """
    _instance = None
    _skills: Dict[str, AutonomousSkill] = {}

    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
        return cls._instance

    def register(self, skill: AutonomousSkill) -> None:
        self._skills[skill.name] = skill
        logger.debug(f"Skill 注册: {skill.name}")

    def get(self, skill_name: str) -> AutonomousSkill | None:
        return self._skills.get(skill_name)

    def list_skills(self) -> List[str]:
        return list(self._skills.keys())

    def get_all(self) -> List[AutonomousSkill]:
        return list(self._skills.values())


skill_registry = SkillRegistry()


class SkillEventBus:
    def __init__(self, skill_registry: SkillRegistry | None = None):
        self._skill_subscribers: Dict[str, List[str]] = {}
        self._state_providers: Dict[str, str] = {}
        self._event_bus = system_event_bus
        # 依赖注入：通过 SkillRegistry 间接访问 skills，避免循环依赖
        self._skill_registry = skill_registry if skill_registry is not None else SkillRegistry()

    def subscribe(self, event_type: str, skill_name: str) -> None:
        if event_type not in self._skill_subscribers:
            self._skill_subscribers[event_type] = []
        self._skill_subscribers[event_type].append(skill_name)
        logger.debug(f"Skill 订阅事件: {skill_name} -> {event_type}")

    def register_state_provider(self, state_name: str, skill_name: str) -> None:
        self._state_providers[state_name] = skill_name

    async def publish(self, event: Dict) -> None:
        event_type = event.get("type", "")
        subscribers = self._skill_subscribers.get(event_type, [])

        for skill_name in subscribers:
            await self._dispatch_to_skill(skill_name, event)

    async def _dispatch_to_skill(self, skill_name: str, event: Dict) -> None:
        # 通过 SkillRegistry 获取 Skill 实例，避免直接导入 app.skills
        skill = self._skill_registry.get(skill_name)
        if skill is None:
            return

        state = await self._resolve_required_state(skill)
        decision = await skill.should_act(event, state)

        if decision.should_act:
            output = await skill.act(event, decision, state)
            await self._process_output(skill, output)

    async def _resolve_required_state(self, skill) -> Dict:
        state = {}
        for state_name, config in getattr(skill, "required_state", {}).items():
            provider = self._state_providers.get(state_name)
            if provider:
                provider_skill = self._skill_registry.get(provider)
                if provider_skill:
                    state[state_name] = await provider_skill.get_state(config)
        return state

    async def _process_output(self, skill, output: SkillOutput) -> None:
        if output.memory_updates:
            for update in output.memory_updates:
                await self._event_bus.emit("memory_updated", **update)

        if output.messages_to_send:
            from app.chat import chat_api
            for msg in output.messages_to_send:
                await chat_api.create_message(**msg)


skill_event_bus = SkillEventBus()
```

### 3.4 与现有 skill_engine.py 的共存策略

**适配器模式**：不直接替换现有被动技能，而是通过适配器将其包装为 `AutonomousSkill`。

```python
# app/skills/adapters.py（新建）
from app.skills.base import AutonomousSkill, ActDecision, SkillOutput
from app.services.skill_engine import evaluate_action_skills, evaluate_inject_skills


class LegacySkillAdapter(AutonomousSkill):
    """
    遗留技能适配器 — 将现有 action/inject 技能包装为 AutonomousSkill。
    
    过渡期间使用，逐步淘汰。
    """
    def __init__(self, legacy_skill_type: str):
        self.legacy_skill_type = legacy_skill_type
        self.name = f"legacy_{legacy_skill_type}"
        self.description = f"遗留技能适配器: {legacy_skill_type}"

    async def should_act(self, event: Dict, state: Dict) -> ActDecision:
        db = state.get("db")
        agent = state.get("agent")
        group_id = event.get("group_id")

        if self.legacy_skill_type == "action":
            result = await evaluate_action_skills(db, agent, group_id, context=event)
            return ActDecision(
                should_act=result.delay_seconds > 0 or result.show_typing,
                priority=10,
                action_type="internal",
                reason="Legacy action skill triggered",
            )
        elif self.legacy_skill_type == "inject":
            prompts = await evaluate_inject_skills(db, agent, group_id, context=event)
            return ActDecision(
                should_act=len(prompts) > 0,
                priority=5,
                action_type="internal",
                reason="Legacy inject skill triggered",
            )
        return ActDecision(should_act=False)

    async def act(self, event: Dict, decision: ActDecision, state: Dict) -> SkillOutput:
        return SkillOutput()
```

```python
logger = logging.getLogger(__name__)


class TriggerEngine:
    async def register_trigger(self, db: AsyncSession, agent_id: int, trigger_data: dict) -> AgentTrigger:
        trigger = AgentTrigger(
            agent_id=agent_id,
            trigger_type=trigger_data["trigger_type"],
            task=trigger_data["task"],
            condition=trigger_data.get("condition", {}),
            expires_at=trigger_data.get("expires_at"),
            max_fires=trigger_data.get("max_fires", -1),
        )
        db.add(trigger)
        await db.commit()
        await db.refresh(trigger)
        return trigger

    async def unregister_trigger(self, db: AsyncSession, agent_id: int, trigger_id: int) -> None:
        result = await db.execute(
            select(AgentTrigger).where(
                AgentTrigger.id == trigger_id,
                AgentTrigger.agent_id == agent_id,
            )
        )
        trigger = result.scalar_one_or_none()
        if trigger:
            await db.delete(trigger)
            await db.commit()

    async def check_triggers(self, db: AsyncSession, event: dict) -> list[AgentTrigger]:
        """检查哪些触发器匹配当前事件"""
        matched = []
        event_type = event.get("type")
        
        result = await db.execute(
            select(AgentTrigger).where(
                AgentTrigger.status == "pending",
            )
        )
        triggers = result.scalars().all()

        for trigger in triggers:
            if await self._matches(trigger, event):
                matched.append(trigger)
        
        return matched

    async def _matches(self, trigger: AgentTrigger, event: dict) -> bool:
        """检查单个触发器是否匹配"""
        t_type = trigger.trigger_type
        
        if t_type == "time":
            return self._match_time(trigger, event)
        elif t_type == "event":
            return self._match_event(trigger, event)
        elif t_type == "semantic":
            return await self._match_semantic(trigger, event)
        elif t_type == "relational":
            return self._match_relational(trigger, event)
        elif t_type == "state":
            return self._match_state(trigger, event)
        elif t_type == "composite":
            return await self._match_composite(trigger, event)
        
        return False

    async def fire_trigger(self, db: AsyncSession, trigger: AgentTrigger) -> None:
        """触发触发器"""
        trigger.fire_count += 1
        
        if trigger.max_fires > 0 and trigger.fire_count >= trigger.max_fires:
            trigger.status = "cancelled"
        else:
            trigger.status = "fired"
            
        await db.commit()
        
        await skill_event_bus.publish({
            "type": "trigger_fired",
            "agent_id": trigger.agent_id,
            "trigger_id": trigger.id,
            "task": trigger.task,
        })


trigger_engine = TriggerEngine()
```

## 十、tools/ 与 skills/ 关系说明

### 10.1 架构分层

```
┌─────────────────────────────────────────────┐
│              Skills 层（决策层）              │
│  AutonomousSkill / AppSkill / StateSkill    │
│  - 感知事件 → 决策 → 调用工具执行              │
├─────────────────────────────────────────────┤
│              Tools 层（执行层）               │
│  ToolPlugin / ToolRegistry                  │
│  - LLM 调用的执行原语，无决策能力              │
├─────────────────────────────────────────────┤
│              ChatApi 层（接口层）             │
│  ChatApi / ConnectionManager                │
│  - AI 与聊天世界的唯一契约边界                 │
└─────────────────────────────────────────────┘
```

### 10.2 核心区别

| 维度 | Tools（工具层） | Skills（技能层） |
|------|---------------|-----------------|
| **角色** | LLM 调用的执行原语 | 自治决策单元 |
| **能力** | 单一动作执行 | 感知 → 决策 → 执行 |
| **状态** | 无状态 | 有状态（State Skill） |
| **触发** | 被 LLM 调用 | 事件驱动、自主触发 |
| **示例** | `send_message`, `store_memory` | `MemorySkill`, `ChatChainSkill` |

### 10.3 调用关系

```
Skill 执行流程：
  should_act(event, state) → 决策是否行动
      ↓
  act(event, decision, state) → 执行动作
      ↓
  ├── 调用 ChatApi 操作聊天世界（发消息、查消息等）
  ├── 调用 Tools 执行 LLM 工具（store_memory、web_search 等）
  └── 更新 State Skill 状态
```

### 10.4 依赖方向约定

**单向依赖原则**：

```
skills → services → chat → database
     ↖                    ↗
      └────── tools ──────┘
```

| 层级 | 允许依赖 | 禁止依赖 |
|------|---------|---------|
| `skills/` | `services/`（通过 SkillRegistry）、`tools/`、`chat/` | 无 |
| `services/` | `chat/`、`database`、`SkillRegistry` | `skills/`（禁止直接导入） |
| `tools/` | `database`、`chat/` | `skills/`、`services/` |
| `chat/` | `database` | `skills/`、`services/`、`tools/` |

**关键规则**：
- `services/` 层必须通过 `SkillRegistry` 间接访问 `skills/`，禁止 `from app.skills import xxx`
- `skills/` 层可以直接导入 `services/`、`tools/`、`chat/`
- `tools/` 层作为执行原语，不依赖任何上层模块

**循环依赖解决方案**：
- `SkillEventBus` 通过构造函数注入 `SkillRegistry`，而非直接导入 `app.skills`
- `SkillRegistry` 在启动时由 `skills/__init__.py` 注册所有 Skill 实例
- 运行时 `SkillEventBus` 通过 `SkillRegistry.get()` 获取 Skill 实例

### 11.7 影子模式验证方案

在 Phase 2-4 期间，采用影子模式验证：新旧系统并行执行，比对关键决策字段。

```python
# app/services/skill_event_bus.py（影子模式）
def _compare_decisions(old_decision: Dict, new_decision: Dict) -> Dict[str, list]:
    """
    比对新旧系统的决策结果，返回差异字段。
    比对关键字段：should_act, priority, action_type, reason
    """
    diffs = {}
    
    compare_fields = ["should_act", "priority", "action_type", "reason"]
    for field in compare_fields:
        old_val = old_decision.get(field)
        new_val = new_decision.get(field)
        
        if old_val != new_val:
            diffs[field] = [old_val, new_val]
    
    return diffs


async def _dispatch_to_skill(self, skill_name: str, event: Dict) -> None:
    skill = self._skill_registry.get(skill_name)
    if skill is None:
        return

    if settings.SHADOW_MODE_ENABLED:
        # 并行执行新旧系统
        old_result = await _run_legacy_handler(skill_name, event)
        new_result = await _run_autonomous_skill(skill, event)
        
        # 结构化字段比对
        old_decision = {
            "should_act": old_result.get("should_act", False),
            "priority": old_result.get("priority", 0),
            "action_type": old_result.get("action_type", "silent"),
            "reason": old_result.get("reason", ""),
        }
        
        new_decision = {
            "should_act": new_result.should_act if hasattr(new_result, "should_act") else False,
            "priority": new_result.priority if hasattr(new_result, "priority") else 0,
            "action_type": new_result.action_type if hasattr(new_result, "action_type") else "silent",
            "reason": new_result.reason if hasattr(new_result, "reason") else "",
        }
        
        diffs = _compare_decisions(old_decision, new_decision)
        if diffs:
            logger.warning(f"🔍 Shadow mode mismatch for {skill_name}:")
            for field, [old_val, new_val] in diffs.items():
                logger.warning(f"  {field}: old={old_val!r} vs new={new_val!r}")
            
        return new_result
    
    return await _run_autonomous_skill(skill, event)
```

## 十二、关键风险与应对

### 12.1 风险清单

| 风险 | 概率 | 影响 | 应对策略 |
|------|------|------|---------|
| AutonomousSkill 设计不当，导致后续扩展困难 | ⭐⭐⭐ | 高 | 先做最小可行接口，预留扩展点 |
| 去中心化后调试困难 | ⭐⭐ | 中 | 声明式依赖让输入输出明确，完善日志 |
| 资源竞争导致死锁 | ⭐⭐ | 中 | 资源调度器使用标准死锁预防策略 |
| 状态同步不一致 | ⭐⭐ | 中 | State Skill 唯一真实来源 + 事件溯源 |
| 复杂度失控 | ⭐⭐⭐ | 高 | 渐进式迁移，每步验证收益和代价 |

### 9.2 复杂度控制策略

1. **单一职责**：每个 Skill 只做一件事
2. **声明式配置**：减少隐式依赖
3. **渐进式迁移**：每阶段完成后验证
4. **可回退设计**：每个阶段都有 Feature Flag
5. **持续重构**：迁移过程中清理旧代码
